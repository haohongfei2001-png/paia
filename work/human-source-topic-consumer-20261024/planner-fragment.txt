// One original Input selection: its creation/deduplication, all placements and
// receipt are planned together and committed by the same original transaction.
async function inputTopicReadSet(store,core,t,r){return {inventory:await humanInventory(store,core,t),projection:await inputProjection(store,t,r.id)};}
function futureInputTopicProjection(g,id){const e=g.state.sources.find(x=>x.input.id===id);if(!e)return null;const b=e.input;return {inputId:id,documentId:b.documentId,contentRevision:e.inputState.contentRevision,lastRemovalSequence:e.inputState.lastRemovalSequence||0,body:b.libraryText??e.source.originalText,note:b.note||'',sourceRecordIds:[e.source.id],identities:[e.source.sourceKey],block:clone(b)};}
export async function prepareHumanInputTopicsPlan(store,core,request,entry){
 keys(request,['operationId','kind','id','expectedRevision','span','topicIds'],['operationId','kind','id','expectedRevision','topicIds']);if(request.kind!=='input'||!idOK(request.operationId)||request.operationId.length<8||!idOK(request.id)||!revisionOK(request.expectedRevision)||!Array.isArray(request.topicIds)||!request.topicIds.length||request.topicIds.length>20||new Set(request.topicIds).size!==request.topicIds.length||request.topicIds.some(x=>!idOK(x)))fail('BNS_HUMAN_REQUEST');
 const g=humanFutureGraph(entry,store,core),read=g?{inventory:clone(g.state),projection:futureInputTopicProjection(g,request.id)}:await store.run(()=>core.transaction(false,t=>inputTopicReadSet(store,core,t,request))),inv=read.inventory,projection=read.projection;
 if(!equal(inv.base,entry)||entry.library?.sealed)fail('BNS_HUMAN_CHANGED');const requestDigest=await hashText(JSON.stringify(request)),prior=inv.operationReceipts.find(x=>x.id===request.operationId);if(prior){if(prior.digest!==requestDigest)fail('BNS_HUMAN_OPERATION_COLLISION');return {duplicate:true,result:clone(prior.result)};}
 if(!projection||projection.block.revision!==request.expectedRevision)return {conflict:true,sourceChanged:true};const span=request.span||{start:0,end:projection.body.length};if(!validInputTopicSpan(projection.body,span))fail('BNS_HUMAN_REQUEST');
 const evidence=g?[await planEvidenceProjection(projection,{role:'primary',selectedFields:['body']},entry.secret,inv.epoch)]:await store.evidenceFor([{inputId:request.id,role:'primary',selectedFields:['body']}]);await humanSourceEvidence(entry,store,core,evidence);
 const body=projection.body.slice(span.start,span.end),full=span.start===0&&span.end===projection.body.length,internal={operationId:request.operationId,actor:'user',body,type:'idea',formation:'explicit',evidence};
 const eligible=inv.provenance.filter(x=>x.inputId===request.id).sort((a,b)=>store.repository.factory.cmp([a.inputId,a.basedOnContentRevision,a.id],[b.inputId,b.basedOnContentRevision,b.id]));let old=null;
 for(const ref of eligible){if(ref.role==='context_only'||ref.contributionType!=='exact_excerpt'||ref.basedOnContentRevision!==projection.contentRevision||ref.span?.start!==span.start||ref.span?.end!==span.end)continue;const row=inv.thoughts.find(x=>x.id===ref.ownerId);if(row?.lifecycle==='active'&&row.thoughtText===body){old=row;break;}}
 if(old&&(old.provenanceType!=='input_original'||old.origin!=='user'))fail('BNS_HUMAN_UNSUPPORTED');
 const allocation=prepareHumanAllocation(store,entry),exactSignature=await keyedHash(entry.secret,['body','idea',body]);let thoughtSequence=entry.sequence?.value||0,revisionSequence=entry.revision?.value||0,row,provenance,dependencies;
 const history=[],placements=clone(inv.placements),states=new Map(inv.topics.map(x=>[x.id,{row:clone(x),index:clone(inv.indices.find(i=>i.id==='thought-read-index:v1:topic:'+x.id)??null)}])),touched=new Set();
 const record=data=>{const prior=inv.revisions.filter(x=>x.entityKey===data.kind+':'+data.entityId);const h=reserveImportantHistory(data,prior,allocation,++revisionSequence);history.push(...h);};
 if(old){row=clone(old);const proof=await sourceEntryProof(store,core,row,entry);provenance=proof.rows.provenance;dependencies=proof.rows.dependencies;}
 else{const id=allocation.uuid(),at=allocation.clock();row=planHumanEntryCreation(internal,{id,at,sequence:++thoughtSequence,exactSignature,evidence,nonContext:evidence});const generationId=allocation.uuid(),pair=planHumanEntryEvidence(internal,evidence[0],{id:allocation.uuid(),generationId,entryId:id,at,generator:null});provenance=[pair.provenance];dependencies=[pair.dependency];row.provenanceType='input_original';applyBinding(row,full?{bodyBinding:'input',workingInputId:request.id,bindingRevision:projection.contentRevision,bindingLength:projection.body.length}:{bodyBinding:'thought'});provenance[0].contributionType='exact_excerpt';provenance[0].span={field:'body',...span,full};}
 const changed=[];
 for(const topicId of request.topicIds){const state=states.get(topicId),topic=state?.row;if(!topic||topic.createdBy!=='user'||topic.lifecycle!=='active'||topic.redirectTo||topic.layoutJobId||topic.sourceRecordIds?.length)fail('BNS_HUMAN_UNSUPPORTED');const id=JSON.stringify([topic.id,topic.activeLayoutGeneration,row.id]),previous=placements.find(x=>x.id===id)??null;if(previous?.lifecycle==='active')continue;
 const section=inv.sections.find(x=>x.id===JSON.stringify([topic.id,topic.activeLayoutGeneration,topic.defaultSectionId]));if(!section||section.lifecycle!=='active'||section.redirectTo||section.sourceRecordIds?.length)fail('BNS_HUMAN_UNSUPPORTED');const last=placements.filter(x=>x.topicId===topic.id&&x.layoutGeneration===topic.activeLayoutGeneration&&x.sectionId===section.sectionId&&x.activeKey===0).sort((a,b)=>store.repository.factory.cmp(LIBRARY_INDEXES.placements.bySectionOrder.map(k=>a[k]),LIBRARY_INDEXES.placements.bySectionOrder.map(k=>b[k]))).at(-1)??null;
 const placement=planHumanPlacementRow(topic,row,previous,section,humanPlacementRank(last,previous,undefined),undefined),r={entryId:row.id,topicId:topic.id,expectedEntryRevision:row.revision,expectedTopicRevision:topic.organizationRevision,...(previous?{expectedPlacementRevision:previous.revision}:{}),operationId:request.operationId};if(!g)await store.run(()=>core.transaction(false,t=>assertMemoryPlacementChangeAllowed(t,topic.id,previous,placement)));
 const planned=planHumanPlacement({entry:row,topic,placement},r,allocation.clock());row=planned.entry;state.row=planned.topic;record({kind:'placement',entityId:id,documentId:topic.id,before:previous,after:placement,fieldMask:['membership','section','order'],actor:'user',reason:'place',important:true,operationId:request.operationId,sourceRecordIds:clone(row.sourceRecordIds)});const at=placements.findIndex(x=>x.id===id);if(at<0)placements.push(placement);else placements[at]=placement;changed.push(placement);Object.assign(state,reserveTouch(state.row,state.index,inv.epoch,allocation,++thoughtSequence));touched.add(topic.id);
 }
 if(!old){const snapshot=entrySnapshot(row);record({kind:'library_entry',entityId:row.id,before:snapshot,after:snapshot,fieldMask:ENTRY_FIELDS,actor:'user',reason:'baseline',important:true,operationId:request.operationId,baseRevision:0,afterRevision:0,sourceRecordIds:clone(row.sourceRecordIds)});}
 row.searchVersion=ownerVersion('entry',row);
 for(const placement of placements.filter(x=>x.entryId===row.id).sort((a,b)=>store.repository.factory.cmp(LIBRARY_INDEXES.placements.byEntry.map(k=>a[k]),LIBRARY_INDEXES.placements.byEntry.map(k=>b[k])))){const state=states.get(placement.topicId);if(!state||state.row.createdBy!=='user'||state.row.sourceRecordIds?.length)fail('BNS_HUMAN_UNSUPPORTED');if(state.row.activeLayoutGeneration===placement.layoutGeneration){Object.assign(state,reserveTouch(state.row,state.index,inv.epoch,allocation,++thoughtSequence));touched.add(placement.topicId);}}
 const result={id:row.id,revision:row.revision},operationReceipt={id:request.operationId,namespace:'thought-library',schemaVersion:1,ownerId:row.id,operationSequence:++thoughtSequence,createdAt:allocation.clock(),digest:requestDigest,result},cap=Object.freeze({});
 const link=sourceLinkedDescriptor(provenance[0],dependencies[0]),journalRows=[{type:'entry',before:old,after:row},{type:'sourceLink',before:old?link:null,after:link},...[...touched].map(id=>({type:'topic',before:inv.topics.find(x=>x.id===id),after:states.get(id).row})),...changed.map(x=>({type:'placement',before:inv.placements.find(p=>p.id===x.id)??null,after:x})),...history.map(x=>({type:'history',before:inv.revisions.find(h=>h.id===x.id)??null,after:x}))];
 plans.set(cap,{kind:'source-add-to-topics',sourceBound:true,store,core,entry:clone(entry),before:{read},request:clone(request),projection,evidence,row,provenance,dependencies,old,history,placements:changed,touchedTopics:[...touched].map(id=>states.get(id).row),touchedIndices:[...touched].map(id=>({id,index:states.get(id).index})),result,operationReceipt,thoughtSequence,revisionSequence,journalRows,allocation:allocation.seal(),events:allocation.events()});return cap;
}
export async function requireHumanInputTopicsPlan(t,cap,store,core){const p=plans.get(cap);if(p?.kind!=='source-add-to-topics'||p.store!==store||p.core!==core)fail('BNS_HUMAN_PLAN_REQUIRED');if(!equal(await inputTopicReadSet(store,core,t,p.request),p.before.read))fail('BNS_HUMAN_CHANGED');await checkEvidenceInTransaction(store,t,p.evidence);return p.allocation;}
